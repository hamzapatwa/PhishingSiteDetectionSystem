@st.cache_data
def load_data():
    """
    Load the CSV data into a DataFrame. Adjust path if needed.
    """
    df = pd.read_csv('urldata.csv')
    return df


@st.cache_resource
def load_pretrained_models(file_path="trained_models.pkl"):
    """
    Load pretrained models (and their metadata) from disk, then store in session state.
    We do a mini conversion step so we can re-generate confusion matrices for this session’s test split.
    """
    if "model_results" not in st.session_state:
        with open(file_path, 'rb') as f:
            st.session_state["model_results"] = pickle.load(f)
    return st.session_state["model_results"]


def plot_conf_matrix(y_true, y_pred, labels=("Legit", "Phish"), title="Confusion Matrix"):
    """
    Generate a Plotly confusion matrix chart.
    """
    cm = confusion_matrix(y_true, y_pred)
    cm_df = pd.DataFrame(cm, index=labels, columns=labels)
    fig = px.imshow(
        cm_df,
        text_auto=True,
        color_continuous_scale="Blues",
        title=title,
        aspect="auto"
    )
    fig.update_xaxes(title="Predicted")
    fig.update_yaxes(title="Actual")
    return fig


def store_model_result(name, model, y_pred_train, y_pred_test, y_train, y_test):
    """
    Store model metrics and results in a dictionary so we can track them collectively.
    """
    acc_train = accuracy_score(y_train, y_pred_train)
    acc_test = accuracy_score(y_test, y_pred_test)
    prec_train = precision_score(y_train, y_pred_train)
    prec_test = precision_score(y_test, y_pred_test)
    cm_fig = plot_conf_matrix(y_test, y_pred_test, title=f"{name} - Confusion Matrix")

    return {
        "model_name": name,
        "model_object": model,
        "acc_train": acc_train,
        "acc_test": acc_test,
        "prec_train": prec_train,
        "prec_test": prec_test,
        "conf_matrix_fig": cm_fig
    }


def train_decision_tree_model(X_train, y_train, X_test, y_test, max_depth):
    tree = DecisionTreeClassifier(max_depth=max_depth, random_state=42)
    tree.fit(X_train, y_train)
    return store_model_result(
        "Decision Tree",
        tree,
        tree.predict(X_train),
        tree.predict(X_test),
        y_train,
        y_test
    )


def train_random_forest_model(X_train, y_train, X_test, y_test, max_depth, n_estimators):
    forest = RandomForestClassifier(max_depth=max_depth, n_estimators=n_estimators, random_state=42)
    forest.fit(X_train, y_train)
    return store_model_result(
        "Random Forest",
        forest,
        forest.predict(X_train),
        forest.predict(X_test),
        y_train,
        y_test
    )


def train_mlp_model(X_train, y_train, X_test, y_test, alpha, hidden_layer_sizes):
    mlp = MLPClassifier(alpha=alpha, hidden_layer_sizes=hidden_layer_sizes, random_state=42)
    mlp.fit(X_train, y_train)
    return store_model_result(
        "Multilayer Perceptrons",
        mlp,
        mlp.predict(X_train),
        mlp.predict(X_test),
        y_train,
        y_test
    )


def train_xgboost_model(X_train, y_train, X_test, y_test, learning_rate, max_depth, n_estimators):
    xgb = XGBClassifier(learning_rate=learning_rate, max_depth=max_depth, n_estimators=n_estimators, random_state=42)
    xgb.fit(X_train, y_train)
    return store_model_result(
        "XGBoost",
        xgb,
        xgb.predict(X_train),
        xgb.predict(X_test),
        y_train,
        y_test
    )


def train_all_models(
        X_train, y_train, X_test, y_test,
        dt_max_depth, rf_max_depth, rf_n_estimators,
        mlp_alpha, mlp_hidden_layer_sizes,
        xgb_learning_rate, xgb_max_depth, xgb_n_estimators
):
    """
    Train all models with provided hyperparameters, storing the results in a dictionary.
    """
    results = {}
    results["Decision Tree"] = train_decision_tree_model(
        X_train, y_train, X_test, y_test, dt_max_depth
    )
    results["Random Forest"] = train_random_forest_model(
        X_train, y_train, X_test, y_test, rf_max_depth, rf_n_estimators
    )
    results["Multilayer Perceptrons"] = train_mlp_model(
        X_train, y_train, X_test, y_test, mlp_alpha, mlp_hidden_layer_sizes
    )
    results["XGBoost"] = train_xgboost_model(
        X_train, y_train, X_test, y_test, xgb_learning_rate, xgb_max_depth, xgb_n_estimators
    )
    return results


def get_tree_graph(tree_model, feature_names):
    """
    Generate a Graphviz tree for a Decision Tree model.
    """
    dot_data = export_graphviz(
        tree_model,
        out_file=None,
        feature_names=feature_names,
        class_names=["Legit", "Phish"],
        filled=True,
        rounded=True,
        special_characters=True
    )
    return graphviz.Source(dot_data)

def calculate_entropy(s):
    if not s:
        return 0
    probs = [float(s.count(c)) / len(s) for c in dict.fromkeys(s)]
    return -sum(p * math.log(p, 2) for p in probs)


def count_subdomains(domain):
    tokens = domain.split('.')
    return max(0, len(tokens) - 2)


def count_digits(url):
    return sum(c.isdigit() for c in url)


def count_special_chars(url):
    return sum(1 for c in url if not c.isalnum() and c not in ".:/?-_")


def uppercase_ratio(url):
    if len(url) == 0:
        return 0
    return sum(1 for c in url if c.isupper()) / len(url)


def get_domain_age(domain):
    """
    Attempt to gather domain creation date from WHOIS data, returning age in days.
    If any error or missing creation date, returns -1.
    """
    try:
        import whois
        w = whois.whois(domain)
        creation_date = w.creation_date
        if isinstance(creation_date, list):
            creation_date = creation_date[0]
        if creation_date is None:
            return -1
        age_days = (datetime.now() - creation_date).days
        return age_days
    except Exception:
        return -1


def get_http_response(url):
    """
    Attempt to get the HTTP response with a short timeout. Return "" on failure.
    """
    try:
        resp = requests.get(url, timeout=5)
        return resp
    except RequestException:
        return ""


def iframe_check(response):
    if not response:
        return 1
    return 0 if re.search(r"<iframe|<frameBorder", response.text, re.IGNORECASE) else 1


def mouse_over_check(response):
    if not response:
        return 1
    return 1 if re.search(r"<script.*onmouseover.*</script>", response.text, re.IGNORECASE | re.DOTALL) else 0


def right_click_check(response):
    if not response:
        return 1
    return 0 if re.search(r"event\.button\s*==\s*2", response.text) else 1


def forwarding_check(response):
    if not response:
        return 1
    return 1 if len(response.history) > 2 else 0


def is_ip_address(url):
    """
    Check if the input is purely an IP address.
    """
    try:
        ipaddress.ip_address(url)
        return True
    except:
        return False


def convert_features_to_numeric(feat_list):
    """
    Convert any string-based features to floats if possible,
    or set them to 0.0 if conversion fails.
    """
    # The first item in feat_list is 'domain' (string),
    # so we skip it. The rest we do a float cast if possible:
    if isinstance(feat_list[0], str):
        feat_list = feat_list[1:]
    return [float(val) if not isinstance(val, str) else 0.0 for val in feat_list]


def extract_features(url: str) -> list:
    """
    Extract advanced lexical and WHOIS-based features for a single URL.
    Returns a list of numeric values in the correct order to pass into the model.
    """
    parsed = urlparse(url)
    domain = parsed.netloc
    if domain.startswith("www."):
        domain = domain.replace("www.", "")

    ip_feat = 1 if is_ip_address(url) else 0
    at_feat = 1 if "@" in url else 0
    length_feat = 1 if len(url) >= 54 else 0
    depth_feat = sum(1 for part in parsed.path.split('/') if part)
    redirect_feat = 1 if url.rfind('//') > 7 else 0
    https_in_domain_feat = 1 if 'https' in domain else 0

    # Check for well-known shorteners
    short_regex = (
        r"bit\.ly|goo\.gl|shorte\.st|go2l\.ink|x\.co|ow\.ly|t\.co|tinyurl|"
        r"tr\.im|is\.gd|cli\.gs|yfrog\.com|migre\.me|ff\.im|tiny\.cc|url4\.eu|"
        r"twit\.ac|su\.pr|twurl\.nl|snipurl\.com|short\.to|BudURL\.com|ping\.fm|"
        r"post\.ly|Just\.as|bkite\.com|snipr\.com|fic\.kr|loopt\.us|doiop\.com|"
        r"short\.ie|kl\.am|wp\.me|rubyurl\.com|om\.ly|to\.ly|bit\.do|t\.co|lnkd\.in|"
        r"db\.tt|qr\.ae|adf\.ly|goo\.gl|bitly\.com|cur\.lv|tinyurl\.com|ow\.ly|bit\.ly|"
        r"ity\.im|q\.gs|is\.gd|po\.st|bc\.vc|twitthis\.com|u\.to|j\.mp|buzurl\.com|"
        r"cutt\.us|u\.bb|yourls\.org|x\.co|prettylinkpro\.com|scrnch\.me|filoops\.info|"
        r"vzturl\.com|qr\.net|1url\.com|tweez\.me|v\.gd|tr\.im|link\.zip\.net"
    )
    tiny_feat = 1 if re.search(short_regex, url) else 0

    prefix_feat = 1 if '-' in domain else 0
    url_entropy = calculate_entropy(url)
    domain_entropy = calculate_entropy(domain)
    subdomain_ct = count_subdomains(domain)
    digit_ct = count_digits(url)
    special_char_ct = count_special_chars(url)
    uppercase_rat = uppercase_ratio(url)
    domain_age = get_domain_age(domain)
    response = get_http_response(url)
    iframe_feat = iframe_check(response)
    mouse_feat = mouse_over_check(response)
    right_click_feat = right_click_check(response)
    forward_feat = forwarding_check(response)

    features_list = [
        domain,  # (Index 0, string, to be dropped)
        ip_feat,
        at_feat,
        length_feat,
        depth_feat,
        redirect_feat,
        https_in_domain_feat,
        tiny_feat,
        prefix_feat,
        url_entropy,
        domain_entropy,
        subdomain_ct,
        digit_ct,
        special_char_ct,
        uppercase_rat,
        domain_age,
        iframe_feat,
        mouse_feat,
        right_click_feat,
        forward_feat,
    ]

    return convert_features_to_numeric(features_list)

def safe_rerun():
    if hasattr(st, "experimental_rerun"):
        st.experimental_rerun()

        st.markdown("**Checking for Missing Values**")

            like `entropy`. The target column is `Label` (0 = Legit, 1 = Phish).

            chosen_col = st.selectbox("Select a numeric column:", numeric_cols,
                                      help="Pick a numeric column to see how its values are spread.")

                with st.spinner("Training Decision Tree..."):
                    dt_result = train_decision_tree_model(X_train, y_train, X_test, y_test, dt_max_depth)
                    st.session_state.setdefault("model_results", {})["Decision Tree"] = dt_result
                    st.success("Decision Tree trained!")
                    safe_rerun()

        elif model_option == "Random Forest":
            rf_max_depth = st.slider("Max Depth", min_value=1, max_value=100, value=40)
            rf_n_estimators = st.slider("Number of Estimators", min_value=10, max_value=300, value=100)
            if st.button("Train Random Forest"):
                with st.spinner("Training Random Forest..."):
                    rf_result = train_random_forest_model(X_train, y_train, X_test, y_test, rf_max_depth,
                                                          rf_n_estimators)
                    st.session_state.setdefault("model_results", {})["Random Forest"] = rf_result
                    st.success("Random Forest trained!")
                    safe_rerun()

        elif model_option == "MLP":
            mlp_alpha = st.slider("Alpha (Regularization)", min_value=0.0001, max_value=0.01, value=0.001, step=0.0001)
            mlp_hidden_layer_sizes_str = st.text_input("Hidden Layer Sizes (comma-separated)", value="200,200,200")
            try:

                mlp_hidden_layer_sizes = (200, 200, 200)
            if st.button("Train MLP"):
                with st.spinner("Training MLP..."):

                    st.session_state.setdefault("model_results", {})["Multilayer Perceptrons"] = mlp_result
                    st.success("MLP trained!")
                    safe_rerun()

            xgb_max_depth = st.slider("Max Depth", min_value=1, max_value=100, value=42)
            xgb_n_estimators = st.slider("Number of Estimators", min_value=10, max_value=500, value=100)

                with st.spinner("Training XGBoost..."):
                    xgb_result = train_xgboost_model(X_train, y_train, X_test, y_test, xgb_learning_rate, xgb_max_depth,
                                                     xgb_n_estimators)
                    st.session_state.setdefault("model_results", {})["XGBoost"] = xgb_result
                    st.success("XGBoost trained!")
                    safe_rerun()

        # Option to train all at once
        st.markdown("---")
        st.markdown("### Or Train All Models at Once")
        if st.button("Train All Models"):
            with st.spinner("Training all models..."):
                dt_max_depth_def = 20
                rf_max_depth_def = 40

                mlp_alpha_def = 0.001

                xgb_lr_def = 0.4
                xgb_max_depth_def = 42
                xgb_n_est_def = 100

                all_results = train_all_models(
                    X_train, y_train, X_test, y_test,
                    dt_max_depth_def,
                    rf_max_depth_def,
                    rf_n_estimators_def,
                    mlp_alpha_def,
                    mlp_hidden_def,
                    xgb_lr_def,
                    xgb_max_depth_def,
                    xgb_n_est_def
                )
                st.session_state["model_results"] = all_results
                st.success("All models trained!")
                safe_rerun()

        # Display model results if they exist
        if "model_results" in st.session_state:
            st.write("### Current Model Results")
            for name, res in st.session_state["model_results"].items():
                with st.expander(f"{name} Results", expanded=False):
                    st.write(f"**Accuracy (Train):** {res['acc_train']:.3f}")
                    st.write(f"**Accuracy (Test):** {res['acc_test']:.3f}")
                    st.write(f"**Precision (Train):** {res['prec_train']:.3f}")

                    st.plotly_chart(res["conf_matrix_fig"], use_container_width=True)
        else:
            st.info("No models loaded or trained yet. Select a model above to begin.")

    #######################################
    # TAB 4: Compare All Models
    #######################################
    with tabs[3]:
        st.subheader("4. Compare All Models")
        with st.expander("How to Compare Models"):
            st.markdown(
                """
                We look at the train and test metrics side by side, sorted by test accuracy

                    "Test Accuracy": round(v["acc_test"], 3),
                    "Train Precision": round(v["prec_train"], 3),
                    "Test Precision": round(v["prec_test"], 3)
